from pathlib import Path
import json,hashlib
W=Path(__file__).resolve().parents[1];A=W/'acceptance-two-r5';R=W/'SIESTA-REMAG'
read=lambda p:json.loads(p.read_text())
hashfile=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
r=read(A/'closeout.json')
for p,h in r['evidence_sha256'].items():assert hashfile(W/p)==h,p
for p,h in r['budget']['old_ledgers_unchanged'].items():assert hashfile(W/p)==h,p
f=r['frozen_preflight_launcher'];assert hashfile(W/f['path'])==f['sha256']
for arm,audit in r['policy_context_failure']['workspace_audit'].items():
 root=A/'policy-arms'/arm
 actual={str(p.relative_to(root)):hashfile(p) for p in root.rglob('*') if p.is_file()}
 assert actual==audit['files_sha256']
for c in r['bundles']:assert hashfile(A/'bundles'/c['key']/'experiment.json')==c['request_sha256']
assert r['execution']['GPU_launches']==0 and r['execution']['retries']==0
assert r['execution']['policy_ToolHost_calls']==0 and r['execution']['sealed_predictions']==0
assert r['budget']['reserved_cents']==5 and r['budget']['unreserved_cents']==150
assert r['budget']['actual_billed_usd'] is None
assert r['allocations']['app']['lifecycle']['state']=='stopped' and not r['allocations']['account_containers']
assert not (A/'runs').exists() and not (A/'policy-predictions-sealed.json').exists()
assert not (R/'scripts/acceptance_two_preflight_modal.py').exists()
assert (A/'closeout.json').read_bytes()==(W.parent/'outputs/Two-outcome-acceptance-results-2026-11-16.json').read_bytes()
assert (A/'closeout.json').read_bytes()==(R/'tests/evidence/research/finite-tool-acceptance-r5-2026-11-16.json').read_bytes()
state=read(W/'qualification-state.json')
assert state['acceptance_two_r5']['status']==r['status']
assert state['scheduling']['status']=='deleted'
print(json.dumps(dict(status='passed',hash_bound_evidence_files=len(r['evidence_sha256']),policy_workspaces_unchanged=True,registered_bundles_unchanged=2,old_ledgers_unchanged=2,output_equals_repo_evidence=True,new_GPU_launches=0,reserved_cents=5,unreserved_cents=150,owned_app_stopped=True,old_heartbeat_deleted=True)))
