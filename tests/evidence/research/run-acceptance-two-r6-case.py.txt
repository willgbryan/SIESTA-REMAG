"""One new registered outcome; never resumes an old driver or queues a successor."""
from pathlib import Path
from datetime import datetime,timezone
import asyncio,json,os,sys,time,traceback
from siesta_remag.discovery.common import load,save
from siesta_remag.corpus import sha256
from siesta_remag.research.execution import execute_query
from siesta_remag.research.experiments import verify_experiment
from acceptance_six_review import recover,review
W=Path(__file__).resolve().parent;A=W/'acceptance-two-r6'
sequence=['fresh-minus-k12','fresh-plus-k12']
if len(sys.argv)!=2 or sys.argv[1] not in sequence:raise ValueError('Expected exactly one registered case')
key=sys.argv[1];index=sequence.index(key);plan=load(A/'registration.json')
for path,h in load(A/'protected-before.json').items():assert sha256(W.parent/path)==h,path
for path,h in load(A/'query-readiness.json')['files'].items():assert sha256(W/path)==h,path
pf=load(W/'acceptance-two-r5/preflight-driver.json');assert pf['preflight_status']=='passed' and pf['allocation_stopped']
case=next(r for r in plan['bundles'] if r['key']==key)
assert sha256(A/'bundles'/key/'experiment.json')==case['request_sha256'];verify_experiment(A/'bundles'/key)
seal=load(A/'policy-predictions-sealed.json')
assert seal['status']=='sealed' and seal['heldouts_executed']==0
for path,h in seal['files'].items():assert sha256(A/path)==h,path
if index:
    prior=load(A/'case-reviews'/f'{sequence[index-1]}.json')
    assert prior['status']=='reviewed' and prior['health']['health_passed'] and prior['allocation_stopped']
summary=A/'case-reviews'/f'{key}.json';summary.parent.mkdir(exist_ok=True)
assert not summary.exists(),'Existing case intent: never resubmit'
row=dict(case=key,status='starting',started_at=datetime.now(timezone.utc).isoformat(),pid=os.getpid(),allocation_stopped=False,automatic_follow_on=False)
save(summary,row);start=time.monotonic()
try:
    result=execute_query(W,'acceptance-two-r6/execution-config.json',f'acceptance-two-r6/bundles/{key}',f'acceptance-two-r6/runs/{key}',spectra=[])
    row.update(execution_status=result['status'],allocation_stopped=result['allocation_stopped'],app_id=result['app_id']);save(summary,row)
    if (A/'runs'/key/'collected/query/report.json').exists():row['recovery']=asyncio.run(recover(A/'runs'/key,A/'native'/key))
    if result['status']!='measured' or not result['allocation_stopped']:raise ValueError('Controlled execution did not yield an accepted stopped result')
    row['health']=review(A,key);row['status']='reviewed'
except BaseException as exc:
    row.update(status='held',error=dict(type=type(exc).__name__,message=str(exc)));traceback.print_exc()
finally:
    row['submission_through_full_review_seconds']=time.monotonic()-start
    row['usable_result_seconds']=row['submission_through_full_review_seconds'] if row['status']=='reviewed' else None
    row['finished_at']=datetime.now(timezone.utc).isoformat();save(summary,row)
    print(json.dumps({k:v for k,v in row.items() if k not in ('health','recovery')}),flush=True)
if row['status']!='reviewed':raise SystemExit(2)
