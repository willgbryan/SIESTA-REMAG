read=lambda p:json.loads(p.read_text())

def symbols(path):
    result=[]
    for line in run('nm','-n','-S',str(path)).splitlines():
        p=line.split()
        if len(p)==4 and re.fullmatch('[0-9a-fA-F]+',p[0]) and re.fullmatch('[0-9a-fA-F]+',p[1]):
            result.append((int(p[0],16),int(p[1],16),p[3]))
    return result

def call(path,pc):
    # gfortran's backtrace reports the return PC minus one. The x86-64 E8
    # direct-call instruction occupies the five bytes ending at this address.
    for length in [5,6]:
        text=run('objdump','-d',f'--start-address={pc+1-length}',f'--stop-address={pc+1}',str(path))
        if re.search(r'\b(?:e8|ff)\b.*\bcallq?\b',text):return text.strip()
    raise AssertionError(text)

def names(table,offset):
    return [name for begin,size,name in table if begin<=offset<begin+size]

stderr=A/'native/fresh-minus-k12/scalar/stderr.log'
frames={int(i):int(pc,16) for i,pc in re.findall(r'^#(\d+)\s+(0x[0-9a-f]+)',stderr.read_text(),re.M)}
es=symbols(exe);ls=symbols(lib)
start=next(x for x,n,name in es if name=='_start')

assert names(es,frames[17]-ebase)==['__m_remag_cycle_MOD_remag_cycle_run']
prepare=next((x,n) for x,n,name in ls if name=='remag_cycle_prepare')
candidates=[]
for offset in range(prepare[0],sum(prepare)):
    base=frames[16]-offset
    if base%4096:continue
    expected={13:'_ZN5remag12ScfCyclePlan4Impl5buildEv',
              14:'_ZN5remag12ScfCyclePlan4ImplC1ERKNS_13ScfCycleShapeERKNS_16ScfCycleBindingsERKNS_17ScfControlProblemEP11CUstream_st',

              16:'remag_cycle_prepare'}
    if all(name in names(ls,frames[i]-base) for i,name in expected.items()):candidates.append(base)
assert candidates==[0x2b32159a6000],candidates
lbase=candidates[0]
chain=[]
for i in [22,19,18,17,16,15,14,13]:

    offset=frames[i]-base

# Resolve the two critical PLT slots through relocation records, not symbol-name guessing.
links=[(exe,0xdc10,'7c5918','remag_cycle_prepare'),(lib,0x34d20,'3cb560','cudaGraphInstantiate')]
plt=[]
for path,address,slot,callee in links:
    text=run('objdump','-d',f'--start-address={address}',f'--stop-address={address+16}',str(path))
    assert '0x'+slot in text
    relocation=next(line for line in run('objdump','-R',str(path)).splitlines() if line.split() and line.split()[0].lstrip('0')==slot)
    assert callee in relocation
    plt.append(dict(binary=path.name,callee=callee,disassembly=text.strip(),relocation=relocation))

assert '0x34d20' in chain[-1]['call_disassembly']
